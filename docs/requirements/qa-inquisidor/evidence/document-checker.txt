def refs(p):
 c=p['candidate'];groups={k:{x['id'] for x in c[k]} for k in ['raw_inputs','semantic_ir','requirements','falsifiers','claims','sources']}
 allids=[]
 for k in groups:
  allids += [x['id'] for x in c[k]]
 if len(allids)!=len(set(allids)):reject('DUPLICATE_ID',allids)
 def subset(v,target):
  if not set(v)<=target:reject('DANGLING_REFERENCE',sorted(set(v)-target))

 for cl in c['claims']:subset(cl['source_ids'],groups['sources'])
 for q in c['questions']:subset(q['answer_source_ids'],groups['sources'])

def formal(f):
 states=f['states'];vars=f['variables'];keys=sorted(vars)
 def valid(v,complete=False):
  if not set(v)<=set(vars) or (complete and set(v)!=set(vars)):reject('VALUATION',v)
  for k,value in v.items():
   if value not in vars[k]:reject('VALUATION',{k:value})
 valid(f['initial']['values'],True)
 for t in f['transitions']:
  if t['from'] not in states or t['to'] not in states:reject('DANGLING_REFERENCE',t['id'])
  valid(t['effects'])
  for clause in t['guard_dnf']:valid(clause)
  if t['actor']=='OTHER' and t['outcome']!='REJECT':reject('AUTHORIZATION',t['id'])
  if t['outcome']=='REJECT' and (t['from']!=t['to'] or t['effects']):reject('REJECT_MUTATES',t['id'])
 def matching(state,vals,a,e):return [t for t in f['transitions'] if t['from']==state and t['actor']==a and t['event']==e and any(all(vals[k]==v for k,v in g.items()) for g in t['guard_dnf'])]
 # Exhaustive guard coverage/exclusivity even for unreachable boolean valuations.
 total=0
 for state,a,e,values in itertools.product(states,f['actors'],f['events'],list(itertools.product(*(vars[k] for k in keys)))):
  v=dict(zip(keys,values));ts=matching(state,v,a,e);total+=1
  if len(ts)!=1:reject('COVERAGE' if not ts else 'NONDETERMINISM',{'state':state,'actor':a,'event':e,'values':v,'matching':[t['id'] for t in ts]})
 def key(state,v):return state,tuple(v[k] for k in keys)
 first=key(f['initial']['state'],f['initial']['values']);seen={first};q=collections.deque([(first,[])]);edge_count=0;done_trace=None;invalidations=0
 while q:
  (state,values),trace=q.popleft();v=dict(zip(keys,values))
  for inv in f['invariants']:
   for pattern in inv['forbidden_patterns']:
    if pattern['state']==state and all(v[k]==value for k,value in pattern['values'].items()):reject('INVARIANT',{'invariant':inv['id'],'trace':trace,'state':state,'values':v})
  if state=='DONE' and done_trace is None:done_trace=trace
  for a,e in itertools.product(f['actors'],f['events']):
   t=matching(state,v,a,e)[0];w={**v,**t['effects']};edge_count+=1
   if a=='QA' and e=='INVALIDATE':
    if t['to']!='ANALYSIS' or w['scope_complete'] or w['evidence_complete']:reject('INVALIDATION',t['id'])
    invalidations+=1
   nxt=key(t['to'],w)
   if nxt not in seen:seen.add(nxt);q.append((nxt,trace+[t['id']]))
 if set(s for s,_ in seen)!=set(states):reject('REACHABILITY',sorted(set(states)-{s for s,_ in seen}))
 if done_trace is None:reject('PROGRESS','No finite path to DONE')
 return {'all_configuration_inputs_checked':total,'reachable_configurations':len(seen),'reachable_edges_checked':edge_count,'finite_completion_witness':done_trace,'invalidation_edges_checked':invalidations,'limitations':'Finite control model only; external phase oracles and real application behavior are not proved.'}
def fresh(report):
 for category,items,key in [('claims',report['ANALYSIS']['claims'],'id'),('components',report['ANALYSIS']['system_inventory'],'component_id'),('evidence',report['VERIFICATION']['evidence'],'id'),('cases',report['TEST PLAN']['cases'],'id'),('results',report['VERIFICATION']['results'],'test_id')]:
  ids=[item[key] for item in items]
  if len(ids)!=len(set(ids)):reject('DUPLICATE_ID',{'category':category,'ids':ids})
 for e in report['VERIFICATION']['evidence']:
  if e['snapshot_sha256']!=report['target_snapshot_sha256']:reject('STALE_EVIDENCE',{'evidence':e['id'],'evidence_snapshot':e['snapshot_sha256'],'target_snapshot':report['target_snapshot_sha256']})
 evidence={e['id'] for e in report['VERIFICATION']['evidence']}
 for cl in report['ANALYSIS']['claims']:
  if not set(cl['evidence_ids'])<=evidence:reject('DANGLING_REFERENCE',cl['id'])
 for x in report['VERIFICATION']['results']:
  if not set(x['evidence_ids'])<=evidence:reject('DANGLING_REFERENCE',x['test_id'])
 for result in report['VERIFICATION']['results']:
  if result['test_id'] not in {t['id'] for t in report['TEST PLAN']['cases']}:reject('DANGLING_REFERENCE',result['test_id'])
 for t in report['TEST PLAN']['cases']:
  if t['id'] not in {x['test_id'] for x in report['VERIFICATION']['results']}:reject('MISSING_RESULT',t['id'])
 if report['VERIFICATION']['verdict']=='PASS_WITHIN_SCOPE' and any(t['authorization_status']=='PENDING' for t in report['TEST PLAN']['cases']):reject('AUTHORIZATION','pending test authorization')
 by_id={e['id']:e for e in report['VERIFICATION']['evidence']}
 by_test={t['id']:t for t in report['TEST PLAN']['cases']}
 for result in report['VERIFICATION']['results']:
  if result['outcome']!='PASS':continue
  planned=by_test[result['test_id']]['execution_kind']
  matches=[by_id[i] for i in result['evidence_ids'] if by_id[i]['kind']==planned and by_id[i]['result']=='PASS']

   matches=[e for e in matches if e['proves_real_execution'] and isinstance(e['exit_code'],int)]
  if not matches:reject('EVIDENCE_KIND_MISMATCH',{'test_id':result['test_id'],'planned':planned,'supporting_kinds':[by_id[i]['kind'] for i in result['evidence_ids']]})
 return 'hashes, cross-references and planned/evidence execution kinds match'
def test(id,expect,fn):

  if detail['reason']!=expect:raise AssertionError((id,expect,detail))
  results.append({'id':id,'status':'PASS','observed':'REJECT','expected_reason':expect,'witness':detail['witness']})
