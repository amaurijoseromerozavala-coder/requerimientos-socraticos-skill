def require(x,msg):

def inspect_fsm(c):
 f=c['fsm'];states=f['states'];actors=f['actors'];events=f['events'];domains=f['variables'];ts=f['transitions']
 require('OTHER' in actors and 'OTHER' in events,'OTHER missing')
 for xs in [states,actors,events,[t['id'] for t in ts]]:require(len(xs)==len(set(xs)),'duplicate ID')
 def val_ok(v,complete=False):
  require(set(v)<=set(domains),'unknown variable')
  if complete:require(set(v)==set(domains),'incomplete valuation')
  for k,vv in v.items():require(vv in domains[k],'out of domain')
 val_ok(f['initial']['values'],True)
 for t in ts:
  require(t['from'] in states and t['to'] in states and t['actor'] in actors and t['event'] in events,'bad reference')
  val_ok(t['effects'])
  for g in t['guard_dnf']:val_ok(g)
  if t['outcome']=='REJECT':require(t['from']==t['to'] and not t['effects'],'reject mutates')
  if t['from'] in f['terminal_states']:require(t['outcome']=='REJECT','terminal allows')
 def key(s,v):return(s,json.dumps(v,sort_keys=True))
 initial=key(f['initial']['state'],f['initial']['values']);seen={initial};queue=collections.deque([(initial,[])])
 explored=0
 while queue:
  (state,serialized),trace=queue.popleft();v=json.loads(serialized)
  for inv in f['invariants']:
   for pattern in inv['forbidden_patterns']:
    require(not(pattern['state']==state and all(v.get(k)==vv for k,vv in pattern['values'].items())),f'invariant witness {trace}')
  for actor,event in itertools.product(actors,events):
   enabled=[t for t in ts if t['from']==state and t['actor']==actor and t['event']==event and any(all(v[k]==vv for k,vv in g.items()) for g in t['guard_dnf'])]
   require(len(enabled)==1,f'coverage/determinism witness {trace+[actor+":"+event]}, matches={len(enabled)}')
   t=enabled[0];explored+=1
   # This is an example-specific oracle, not a parser for arbitrary natural-language logic.
   if t['outcome']=='ALLOW':require((state,actor,event)==('inactive','operator','enable'),'authorization witness '+str(trace+[t['id']]))
   w=dict(v);w.update(t['effects']);n=key(t['to'],w)
   if n not in seen:seen.add(n);queue.append((n,trace+[t['id']]))
 require({s for s,_ in seen}==set(states),'unreachable state')
 return {'reachable_configurations':len(seen),'evaluated_inputs':explored}
